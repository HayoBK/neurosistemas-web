"" -}} {{- if and (gt (len $limpia) 1) (not (in $particulas (lower $limpia))) }} {{- $palabras = $palabras | append $limpia }} {{- end }} {{- end -}} {{- $ini := "" -}} {{- if gt (len $palabras) 0 }} {{- $ini = substr (index $palabras 0) 0 1 -}} {{- if gt (len $palabras) 1 }} {{- $ini = printf "%s%s" $ini (substr (index $palabras (sub (len $palabras) 1)) 0 1) -}} {{- end }} {{- end -}} {{- upper $ini -}}
